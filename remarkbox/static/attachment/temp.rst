Do the following commands from the root pylowiki project directory::

  sudo chown -R www-data:www-data pylowiki/public/attachment
  sudo chmod -R 744 pylowiki/public/attachment

  sudo chown -R www-data:www-data egg-cache
  sudo chmod -R 744 egg-cache

  sudo chown -R www-data:www-data data
  sudo chmod -R 744 data
